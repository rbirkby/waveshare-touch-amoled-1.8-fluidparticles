cmake_minimum_required(VERSION 3.13)
set(PICO_BOARD pico2 CACHE STRING "Board type")
include($ENV{PICO_SDK_PATH}/external/pico_sdk_import.cmake)
project(dispprobe C CXX ASM)
pico_sdk_init()
add_executable(dispprobe main.c)
pico_generate_pio_header(dispprobe ${CMAKE_CURRENT_LIST_DIR}/qspi.pio)
target_link_libraries(dispprobe pico_stdlib hardware_pio hardware_dma hardware_clocks hardware_i2c)
pico_enable_stdio_usb(dispprobe 1)
pico_enable_stdio_uart(dispprobe 0)
pico_add_extra_outputs(dispprobe)
