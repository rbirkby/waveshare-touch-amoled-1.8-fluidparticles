cmake_minimum_required(VERSION 3.13)
include($ENV{PICO_SDK_PATH}/pico_sdk_init.cmake)
project(hwprobe C CXX ASM)
set(CMAKE_C_STANDARD 11)
pico_sdk_init()
add_executable(hwprobe main.c)
target_link_libraries(hwprobe pico_stdlib hardware_i2c)
pico_enable_stdio_usb(hwprobe 1)
pico_enable_stdio_uart(hwprobe 0)
pico_add_extra_outputs(hwprobe)
